SSH (key-only, user "dev"):
{{- if eq .Values.sshServer.service.type "NodePort" }}

  NODE_PORT=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "opencode.fullname" . }}-ssh -o jsonpath='{.spec.ports[0].nodePort}')
  ssh -p $NODE_PORT dev@<any-node-ip>
{{- else }}

  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "opencode.fullname" . }}-ssh {{ .Values.sshServer.port }}:ssh

  Run the TUI inside the pod: ssh -t ... dev@... opencode
